# (red flash, green sweep). Use this to verify the v2 effects renderer.

# Format: <delay-ms> <wire-line>

0     F=N;B=0;S=pre-race

# Green start: bright sweep L→R for ~500 ms then settles to solid green.
4000  F=G;B=0;S=racing

# Yellow progressions: static → single-waved (2 Hz strobe) → double-waved (4 Hz).
4000  F=Y;B=0;S=racing
3000  F=Y;B=1;S=racing
3000  F=Y;B=2;S=racing

# Back to none briefly so the next flag transition is a real onset.
2000  F=N;B=0;S=racing

3000  F=R;B=1;S=racing
